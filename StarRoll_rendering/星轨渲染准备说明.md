## 星轨渲染原理

### 什么是星轨？

星轨是由于地球自转，星星在天空中看起来移动形成的轨迹。

```
        北极星 (不动点)
           ⭐
          /|\
         / | \
        /  |  \
       ⭐  |  ⭐  ← 星星围绕北极星旋转
        \  |  /
         \ | /
          \|/
           
```

### 技术实现方案

#### 方案1: 长曝光模拟（推荐用于拍照）
```typescript
// 保存星星的历史位置
const starTrails: Map<number, Vector3[]> = new Map();

function updateStarTrails(deltaTime: number) {
    stars.forEach(star => {
        // 计算新位置（基于时间推进）
        const newPos = calculateStarPosition(star, currentTime + deltaTime);
        
        // 保存到轨迹数组
        if (!starTrails.has(star.hip)) {
            starTrails.set(star.hip, []);
        }
        starTrails.get(star.hip).push(newPos);
    });
}

function renderStarTrails() {
    starTrails.forEach((positions, hip) => {
        // 绘制连线（轨迹）
        const geometry = new LineGeometry(positions);
        const material = new LineMaterial({
            color: starColor,
            opacity: 0.6,
            transparent: true
        });
        scene.add(new Line(geometry, material));
    });
}
```

## 推荐实现方案

### 为拍照识别优化的星轨方案

考虑到您的需求（拍照识别星星），推荐 **方案1+优化**：

```typescript
class StarTrailRenderer {
    private starPositions: Map<number, StarTrailData> = new Map();
    private trailLines: THREE.Group = new THREE.Group();
    private timeSpeed: number = 1.0;  // 时间流速倍数
    
    // 星轨数据
    interface StarTrailData {
        hip: number;
        positions: THREE.Vector3[];
        colors: THREE.Color[];
        timestamps: number[];
    }
    
    /**
     * 更新星轨（按时间推进）
     */
    updateTrails(deltaTime: number) {
        const timeStep = deltaTime * this.timeSpeed;
        
        this.stars.forEach(star => {
            // 计算新时间的星星位置
            const newTime = this.currentTime + timeStep;
            const newPos = this.calculateStarPositionAtTime(star, newTime);
            
            // 添加到轨迹
            const trail = this.getOrCreateTrail(star.hip);
            trail.positions.push(newPos);
            trail.timestamps.push(newTime);
            
            // 限制轨迹长度（避免内存溢出）
            if (trail.positions.length > 1000) {
                trail.positions.shift();
                trail.timestamps.shift();
            }
        });
        
        // 重新渲染轨迹线
        this.renderTrailLines();
    }
    
    /**
     * 渲染轨迹线
     */
    renderTrailLines() {
        // 清除旧的轨迹
        this.trailLines.clear();
        
        this.starPositions.forEach((trail, hip) => {
            if (trail.positions.length < 2) return;
            
            // 创建轨迹线
            const geometry = new THREE.BufferGeometry();
            geometry.setFromPoints(trail.positions);
            
            // 渐变颜色和透明度
            const colors = this.generateTrailColors(trail);
            geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
            
            const material = new THREE.LineBasicMaterial({
                vertexColors: true,
                transparent: true,
                opacity: 0.8,
                blending: THREE.AdditiveBlending
            });
            
            const line = new THREE.Line(geometry, material);
            this.trailLines.add(line);
        });
        
        this.scene.add(this.trailLines);
    }
    
    /**
     * 生成渐变颜色（轨迹头部亮，尾部暗）
     */
    generateTrailColors(trail: StarTrailData): number[] {
        const colors: number[] = [];
        const count = trail.positions.length;
        
        trail.positions.forEach((_, index) => {
            // 从暗到亮的渐变
            const alpha = index / count;
            const brightness = 0.2 + alpha * 0.8;
            
            colors.push(brightness, brightness, brightness);
        });
        
        return colors;
    }
    
    /**
     * 计算指定时间的星星位置
     */
    calculateStarPositionAtTime(star: StarMeta, time: Date): THREE.Vector3 {
        // 计算该时刻的本地恒星时
        const lst = HorizonCoordinates.calculateLocalSiderealTime(
            this.observerLocation.longitude,
            time
        );
        
        // 转换为地平坐标
        const { altitude, azimuth } = HorizonCoordinates.equatorialToHorizon(
            star.equatorialCoordinate.rightAscension,
            star.equatorialCoordinate.declination,
            this.observerLocation.latitude,
            lst
        );
        
        // 转换为3D位置
        return HorizonCoordinates.horizonToVector3(
            altitude,
            azimuth,
            this.SKY_RADIUS
        );
    }
    
    /**
     * 控制时间流速
     */
    setTimeSpeed(speed: number) {
        this.timeSpeed = speed;  // 1.0 = 正常, 10.0 = 10倍速, 100.0 = 100倍速
    }
    
    /**
     * 清除所有轨迹
     */
    clearTrails() {
        this.starPositions.clear();
        this.trailLines.clear();
    }
    
    /**
     * 导出轨迹数据（用于拍照识别）
     */
    exportTrailData(): StarTrailExport[] {
        const exports: StarTrailExport[] = [];
        
        this.starPositions.forEach((trail, hip) => {
            exports.push({
                hip: hip,
                positions: trail.positions.map(p => ({
                    x: p.x,
                    y: p.y,
                    z: p.z
                })),
                startTime: trail.timestamps[0],
                endTime: trail.timestamps[trail.timestamps.length - 1],
                duration: trail.timestamps[trail.timestamps.length - 1] - trail.timestamps[0]
            });
        });
        
        return exports;
    }
}
```

// 快速生成1小时的星轨
async function generateTrailsForPhoto(duration: number = 3600) {
    // 暂停实时渲染
    pauseRendering();
    
    // 清除现有轨迹
    trailRenderer.clearTrails();
    
    // 设置超高速（1000倍）
    trailRenderer.setTimeSpeed(1000.0);
    
    // 生成轨迹
    const steps = 100;  // 100个步骤
    const stepTime = duration / steps;
    
    for (let i = 0; i < steps; i++) {
        trailRenderer.updateTrails(stepTime);
        
        // 显示进度
        updateProgress(i / steps);
    }
    
    // 恢复正常
    trailRenderer.setTimeSpeed(1.0);
    resumeRendering();
    
    // 导出数据
    const trailData = trailRenderer.exportTrailData();
    return trailData;
}

// 拍照并识别
async function captureWithTrails() {
    // 生成轨迹
    const trails = await generateTrailsForPhoto(3600);  // 1小时
    
    // 拍照
    const image = captureScreen();
    
    // 识别
    const stars = identifyStarsFromTrails(image, trails);
    
    return stars;
}

function animate(dt) {
    trailUpdateTimer += dt;

        trailRenderer.updateTrails(trailUpdateTimer);
        trailUpdateTimer = 0;
