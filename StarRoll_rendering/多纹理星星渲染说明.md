## 代码实现

### 核心算法

```typescript
private createStarFieldFromCatalog(stars: StarMeta[]): void {
    // 步骤1: 按星等分组
    const brightStars = [];  // < 1等
    const mediumStars = [];  // 1-2.5等
    const dimStars = [];     // 2.5-4.5等
    
    stars.forEach(star => {
        // 计算位置和颜色
        const pos = calculatePosition(star);
        const color = this.bvToRGB(star.bvColor);
        
        // 分组
        if (star.magnitude < 1.0) {
            brightStars.push({ pos, color, star });
        } else if (star.magnitude < 2.5) {
            mediumStars.push({ pos, color, star });
        } else {
            dimStars.push({ pos, color, star });
        }
    });
    
    // 步骤2: 为每组创建Points对象
    const starGroup = new THREE.Group();
    
    // 极亮星 - 带光芒纹理
    starGroup.add(this.createStarPoints(
        brightStars, 
        '/texture/star16x16_ray.png', 
        8.0
    ));
    
    // 中等亮星 - 普通纹理
    starGroup.add(this.createStarPoints(
        mediumStars, 
        '/texture/star16x16.png', 
        5.0
    ));
    
    // 暗星 - 普通纹理（更小）
    starGroup.add(this.createStarPoints(
        dimStars, 
        '/texture/star16x16.png', 
        3.0
    ));
    
    this.scene.add(starGroup);
}
```

### 创建星星点对象

```typescript
private createStarPoints(
    starsData: Array<{pos: Vector3, color: Color, star: StarMeta}>,
    texturePath: string,
    baseSize: number
): THREE.Points {
    // 构建几何体
    const positions: number[] = [];
    const colors: number[] = [];
    
    starsData.forEach(({ pos, color }) => {
        positions.push(pos.x, pos.y, pos.z);
        colors.push(color.r, color.g, color.b);
    });
    
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    
    // 加载纹理并创建材质
    const textureLoader = new THREE.TextureLoader();
    const texture = textureLoader.load(texturePath);
    
    const material = new THREE.PointsMaterial({
        size: baseSize,
        map: texture,
        transparent: true,
        opacity: 1.0,
        vertexColors: true,        // 保留星星的真实颜色
        sizeAttenuation: false,    // 固定大小
        blending: THREE.AdditiveBlending,
        depthWrite: false
    });
    
    return new THREE.Points(geometry, material);
}
```

// 获取可见星星的屏幕坐标
function getVisibleStarsWithScreenCoords(): StarScreenData[] {
    const result: StarScreenData[] = [];
    
    this.starGroup.traverse((points) => {
        if (points instanceof THREE.Points) {
            const positions = points.geometry.attributes.position;
            
            for (let i = 0; i < positions.count; i++) {
                const worldPos = new THREE.Vector3(
                    positions.getX(i),
                    positions.getY(i),
                    positions.getZ(i)
                );
                
                // 转换为屏幕坐标
                const screenPos = worldPos.clone().project(this.camera);
                
                result.push({
                    hip: getHIPFromIndex(i),
                    screenX: (screenPos.x + 1) * window.innerWidth / 2,
                    screenY: (-screenPos.y + 1) * window.innerHeight / 2,
                    worldPos: worldPos,
                    magnitude: getMagnitude(i)
                });
            }
        }
    });
    
    return result;
}
